=== lib.sh ===
#!/bin/bash
# Helpers for the bw-870 measurements against a throwaway dev-mode OpenBao 2.6.1 (synthetic values only).
S=$PWD
BAO_PORT=18870
BAO=http://127.0.0.1:$BAO_PORT
# shellcheck disable=SC2034  # read by setup.sh, which sources this file; no measurement uses it
ROOT=bw870-synthetic-root-token
ISS=https://idp.synthetic.invalid

b64u() { openssl base64 -A | tr '+/' '-_' | tr -d '='; }

# jwt <sub> <groups-json-array> <aud-json> <exp-seconds-from-now> [extra-claims-json-object-body]
# prints a signed RS256 JWT from the synthetic IdP key
jwt() {
  local sub=$1 groups=$2 aud=$3 ttl=$4 extra=${5:-}
  local now exp hdr pl sig
  now=$(date +%s)
  exp=$((now + ttl))
  hdr=$(printf '{"alg":"RS256","typ":"JWT"}' | b64u)
  pl=$(printf '{"iss":"%s","sub":"%s","aud":%s,"groups":%s,"iat":%s,"nbf":%s,"exp":%s%s}' \
    "$ISS" "$sub" "$aud" "$groups" "$now" "$now" "$exp" "${extra:+,$extra}" | b64u)
  sig=$(printf '%s.%s' "$hdr" "$pl" | openssl dgst -sha256 -sign "$S/idp.key" | b64u)
  printf '%s.%s.%s' "$hdr" "$pl" "$sig"
}

# api <method> <path-after-/v1/> <token> [json-body] [namespace]
# prints "<http status> <body>" on one line
api() {
  local m=$1 p=$2 t=$3 d=${4:-} ns=${5:-}
  local args=(-s -o /dev/stdout -w '\n%{http_code}' -X "$m" -H "X-Vault-Token: $t")
  [ -n "$ns" ] && args+=(-H "X-Vault-Namespace: $ns")
  [ -n "$d" ] && args+=(-H 'Content-Type: application/json' -d "$d")
  local out
  out=$(curl "${args[@]}" "$BAO/v1/$p")
  printf '%s %s\n' "${out##*$'\n'}" "$(printf '%s' "${out%$'\n'*}" | tr -d '\n' | cut -c1-300)"
}

# login <jwt> [mount] [namespace] -> prints the client token ("" on failure)
login() {
  local j=$1 mount=${2:-jwt} ns=${3:-}
  local args=(-s -X POST -H 'Content-Type: application/json' -d "{\"role\":\"tenant\",\"jwt\":\"$j\"}")
  [ -n "$ns" ] && args+=(-H "X-Vault-Namespace: $ns")
  curl "${args[@]}" "$BAO/v1/auth/$mount/login" | jq -r '.auth.client_token // empty'
}

# status-only of an api call
code() { api "$@" | cut -d' ' -f1; }

#!/bin/bash
# Starts a throwaway dev-mode OpenBao (the pinned image) on loopback and configures synthetic tenants.
# The synthetic dev root token is used here and only here; measurements log in through jwt auth.
set -eu
. ./lib.sh
IMAGE='ghcr.io/openbao/openbao:2.6.1@sha256:5b2486ab0fb90bbc788cc345b0a08616dfb375873ee8be5df3a2fd4d378a67e0'
NAME=bw870-bao

df -m /home | tail -1

if [ ! -f "$S/idp.key" ]; then
  openssl genrsa -out "$S/idp.key" 2048 2>/dev/null
  chmod 600 "$S/idp.key"
  openssl rsa -in "$S/idp.key" -pubout -out "$S/idp.pub" 2>/dev/null
fi

if [ -n "$(docker ps -aq --filter "name=^$NAME\$")" ]; then docker rm -f "$NAME" >/dev/null; fi
docker run -d --rm --name "$NAME" -e BAO_DEV_ROOT_TOKEN_ID="$ROOT" -e BAO_DEV_LISTEN_ADDRESS=0.0.0.0:8200 \
  -p "127.0.0.1:$BAO_PORT:8200" "$IMAGE" >/dev/null

ready=no
for _ in $(seq 30); do
  if curl -sf "$BAO/v1/sys/health" >/dev/null 2>&1; then ready=yes; break; fi
  sleep 1
done
echo "ready=$ready"
[ "$ready" = yes ] || exit 1

echo "version: $(curl -s "$BAO/v1/sys/health" | jq -r '.version, .sealed' | tr '\n' ' ')"

# jwt auth against the synthetic IdP key
api POST sys/auth/jwt "$ROOT" '{"type":"jwt"}'
pem=$(cat "$S/idp.pub")
api POST auth/jwt/config "$ROOT" "$(jq -n --arg p "$pem" --arg i "$ISS" '{jwt_validation_pubkeys:$p,bound_issuer:$i}')"
api PUT auth/jwt/role/tenant "$ROOT" '{"role_type":"jwt","user_claim":"sub","groups_claim":"groups","bound_audiences":["openbao"],"token_ttl":"15m","token_max_ttl":"15m"}'

acc=$(curl -s -H "X-Vault-Token: $ROOT" "$BAO/v1/sys/auth" | jq -r '.data["jwt/"].accessor')
echo "jwt accessor present: $([ -n "$acc" ] && echo yes || echo no)"

# transit and kv
api POST sys/mounts/transit "$ROOT" '{"type":"transit"}'
for t in a b; do
  api POST "transit/keys/$t-artifact" "$ROOT" '{}'
  api PUT "secret/data/tenants/$t/x" "$ROOT" "{\"data\":{\"v\":\"$t-synthetic\"}}"
done

mkpol() {
  local t=$1
  jq -n --arg t "$t" '{policy: ("path \"secret/data/tenants/" + $t + "/*\" {capabilities=[\"create\",\"read\",\"update\"]}\npath \"secret/metadata/tenants/" + $t + "/*\" {capabilities=[\"read\",\"list\"]}\npath \"transit/encrypt/" + $t + "-artifact\" {capabilities=[\"update\"]}\npath \"transit/decrypt/" + $t + "-artifact\" {capabilities=[\"update\"]}\n")}'
}
for t in a b; do
  api PUT "sys/policies/acl/tenant-$t" "$ROOT" "$(mkpol "$t")"
  gid=$(curl -s -X POST -H "X-Vault-Token: $ROOT" -d "{\"name\":\"tenant-$t\",\"type\":\"external\",\"policies\":[\"tenant-$t\"]}" "$BAO/v1/identity/group" | jq -r '.data.id')
  api POST identity/group-alias "$ROOT" "$(jq -n --arg n "tenant-$t" --arg a "$acc" --arg g "$gid" '{name:$n,mount_accessor:$a,canonical_id:$g}')"
done
echo "setup done"

#!/bin/bash
# M5: one OpenBao namespace per tenant (ta, tb), each with its own kv, transit and jwt auth mount.
# Setup steps use the dev root token and are labelled ADMIN; every measurement uses a token from jwt login.
set -u
. ./lib.sh
PUB=$(cat "$S/idp.pub")

echo "ADMIN: create namespaces and per-namespace mounts"
for n in ta tb; do
  g="tenant-${n#t}"
  echo "ns $n create: $(code PUT "sys/namespaces/$n" "$ROOT" '{}')"
  code POST sys/auth/jwt "$ROOT" '{"type":"jwt"}' "$n" >/dev/null
  code POST sys/mounts/secret "$ROOT" '{"type":"kv","options":{"version":"2"}}' "$n" >/dev/null
  code POST sys/mounts/transit "$ROOT" '{"type":"transit"}' "$n" >/dev/null
  code POST transit/keys/artifact "$ROOT" '{}' "$n" >/dev/null
  code PUT sys/policies/acl/t "$ROOT" "$(jq -n '{policy: "path \"secret/data/*\" {capabilities=[\"create\",\"read\",\"update\"]}\npath \"transit/encrypt/artifact\" {capabilities=[\"update\"]}\npath \"transit/decrypt/artifact\" {capabilities=[\"update\"]}\n"}')" "$n" >/dev/null
  code POST auth/jwt/config "$ROOT" "$(jq -n --arg k "$PUB" --arg i "$ISS" '{jwt_validation_pubkeys:$k,bound_issuer:$i}')" "$n" >/dev/null
  code POST auth/jwt/role/tenant "$ROOT" "$(jq -n --arg g "$g" '{role_type:"jwt",user_claim:"sub",bound_audiences:["openbao"],bound_claims:{groups:[$g]},token_policies:["t"],token_ttl:"15m",token_max_ttl:"15m"}')" "$n" >/dev/null
done

ua=$(jwt user-a '["tenant-a"]' '"openbao"' 300)
uab=$(jwt user-ab '["tenant-a","tenant-b"]' '"openbao"' 300)
echo "== login inside the matching namespace"
ta=$(login "$ua" jwt ta)
echo "user a, ns ta: expected token; observed: $([ -n "$ta" ] && echo token || echo refused)"
echo "user a, ns tb: expected refused (bound_claims); observed: $([ -n "$(login "$ua" jwt tb)" ] && echo TOKEN || echo refused)"
echo "user a, root namespace (jwt mount from the earlier setup): observed: $([ -n "$(login "$ua" jwt)" ] && echo token || echo refused)"

echo "== use of the ns-ta token"
echo "write in ta: expected 200; observed: $(code PUT secret/data/x "$ta" '{"data":{"v":"a"}}' ta)"
echo "read in ta: expected 200; observed: $(code GET secret/data/x "$ta" '' ta)"
echo "same token, header namespace tb: expected 403; observed: $(code GET secret/data/x "$ta" '' tb)"
echo "same token, root namespace: expected 403; observed: $(code GET secret/data/tenants/a/x "$ta")"
echo "same token, no header, path ta/ in URL: observed: $(code GET ta/secret/data/x "$ta")"

echo "== transit keys are separate per namespace even with the same name"
